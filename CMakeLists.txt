cmake_minimum_required(VERSION 3.20)
project(homm3_port LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(HOMM3_HARNESS "Build private deterministic parity instrumentation" OFF)
option(HOMM3_GAME "Build the scenario/game client (OFF: menu regression client)" ON)
set(HOMM3_WORKER_SOURCE "$ENV{HOMM3_WORKER_SOURCE}" CACHE PATH "External authority worker directory")
set(HOMM3_WEB_OUTPUT "${CMAKE_SOURCE_DIR}/build/client" CACHE PATH "Browser artifact output directory")
option(HOMM3_SERVER "Build the experimental native authority worker" OFF)
if(HOMM3_SERVER AND (EMSCRIPTEN OR NOT HOMM3_GAME))
  message(FATAL_ERROR "HOMM3_SERVER requires a native full-game build")
endif()
if(HOMM3_GAME)
  set(MANIFEST game-sources)
  set(ENTRY port/game_main.cpp)
else()
  set(MANIFEST sources)
  set(ENTRY port/main.cpp)
endif()
file(READ "${CMAKE_SOURCE_DIR}/port/${MANIFEST}.json" SOURCE_MANIFEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "port/${MANIFEST}.json")
foreach(GROUP core platform frontend)
  string(JSON COUNT LENGTH "${SOURCE_MANIFEST}" "${GROUP}")
  math(EXPR LAST "${COUNT} - 1")
  foreach(I RANGE 0 ${LAST})
    string(JSON UNIT GET "${SOURCE_MANIFEST}" "${GROUP}" ${I})
    if(GROUP STREQUAL "core")
      list(APPEND PORT_SOURCES "src/${UNIT}.cpp")
    else()
      list(APPEND PORT_SOURCES "port/${GROUP}/${UNIT}.cpp")
    endif()
  endforeach()
endforeach()
add_executable(homm3-port ${ENTRY} ${PORT_SOURCES})
if(HOMM3_GAME)
  target_compile_definitions(homm3-port PRIVATE HOMM3_FULL_GAME)
  set_target_properties(homm3-port PROPERTIES OUTPUT_NAME homm3-game)
endif()
if(HOMM3_SERVER)
  if(NOT EXISTS "${HOMM3_WORKER_SOURCE}/main.cpp")
    message(FATAL_ERROR "Set HOMM3_WORKER_SOURCE to the authority worker source directory")
  endif()
  set(WORKER_SOURCES)
  foreach(UNIT main session terminal actions encounters market town_actions map_services guild exchange puzzle combat projection)
    list(APPEND WORKER_SOURCES "${HOMM3_WORKER_SOURCE}/${UNIT}.cpp")
  endforeach()
  set_property(TARGET homm3-port PROPERTY SOURCES ${WORKER_SOURCES} ${PORT_SOURCES})
  target_include_directories(homm3-port PRIVATE "${HOMM3_WORKER_SOURCE}")
  target_compile_definitions(homm3-port PRIVATE HOMM3_SERVER)
  set_target_properties(homm3-port PROPERTIES OUTPUT_NAME homm3-worker)
endif()
if(HOMM3_HARNESS)
  target_sources(homm3-port PRIVATE port/harness/runtime.cpp)
  target_compile_definitions(homm3-port PRIVATE HOMM3_HARNESS)
endif()
target_include_directories(homm3-port PRIVATE port/compat include)
target_compile_definitions(homm3-port PRIVATE HOMM3_PORT SDL_MAIN_HANDLED)
target_compile_options(homm3-port PRIVATE "-include${CMAKE_SOURCE_DIR}/port/compat/prefix.h"
  -fms-extensions -ffunction-sections -fdata-sections -Wno-writable-strings -Wno-deprecated-declarations
  -Wno-ignored-attributes -Wno-unknown-pragmas -Wno-non-c-typedef-for-linkage -Wno-delete-incomplete -Wno-invalid-constexpr -Wno-out-of-line-declaration -Wno-address-of-temporary)
if(EMSCRIPTEN)
  target_compile_options(homm3-port PRIVATE -sUSE_ZLIB=1 -sUSE_SDL=2 -fexceptions)
  target_link_options(homm3-port PRIVATE -g2 -sUSE_ZLIB=1 -sUSE_SDL=2 -fexceptions -sASYNCIFY=1 -sASYNCIFY_STACK_SIZE=1048576
    -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=268435456 -sSTACK_SIZE=8388608 -sFORCE_FILESYSTEM=1
    "-sEXPORTED_RUNTIME_METHODS=['FS','callMain']" -sINVOKE_RUN=0 -sEXIT_RUNTIME=0 -sENVIRONMENT=web)
  if(HOMM3_GAME)
    target_link_options(homm3-port PRIVATE -lidbfs.js "-sEXPORTED_RUNTIME_METHODS=['FS','IDBFS','callMain']")
    set(WEB_NAME game)
  else()
    set(WEB_NAME homm3)
  endif()
  set_target_properties(homm3-port PROPERTIES OUTPUT_NAME ${WEB_NAME} SUFFIX .js
    RUNTIME_OUTPUT_DIRECTORY "${HOMM3_WEB_OUTPUT}")
  if(HOMM3_HARNESS)
    set_target_properties(homm3-port PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/web")
  endif()
else()
  find_package(SDL2 CONFIG REQUIRED)
  find_package(ZLIB REQUIRED)
  target_link_libraries(homm3-port PRIVATE SDL2::SDL2 ZLIB::ZLIB)
  if(APPLE)
    target_link_options(homm3-port PRIVATE -Wl,-dead_strip)
  else()
    target_link_options(homm3-port PRIVATE -Wl,--gc-sections)
  endif()
endif()

if(HOMM3_GAME)
  if(EMSCRIPTEN)
    set(VIDEO_TARGET web)
  else()
    set(VIDEO_TARGET native)
  endif()
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/build-video.py" --target ${VIDEO_TARGET}
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
  set(VIDEO_ROOT "${CMAKE_SOURCE_DIR}/build/ffmpeg-${VIDEO_TARGET}/install")
  target_include_directories(homm3-port PRIVATE "${VIDEO_ROOT}/include")
  foreach(LIB avformat avcodec swscale swresample avutil)
    target_link_libraries(homm3-port PRIVATE "${VIDEO_ROOT}/lib/lib${LIB}.a")
  endforeach()
endif()
